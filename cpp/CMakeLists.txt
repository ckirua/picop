cmake_minimum_required(VERSION 3.16)
project(smh_q VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(SMH_Q_NATIVE_ARCH "Enable -march=native for smh_q lib and benchmarks" OFF)
if(DEFINED ENV{SMH_Q_NATIVE_ARCH} AND "$ENV{SMH_Q_NATIVE_ARCH}" STREQUAL "1")
  set(SMH_Q_NATIVE_ARCH ON CACHE BOOL "Enable -march=native for smh_q lib and benchmarks" FORCE)
endif()

include(CheckIPOSupported)
check_ipo_supported(RESULT smh_q_ipo_supported OUTPUT smh_q_ipo_error)
if(smh_q_ipo_supported)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
add_library(smh_q STATIC src/ring.cpp)
target_include_directories(smh_q PUBLIC include)
if(SMH_Q_NATIVE_ARCH)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(smh_q PRIVATE -march=native)
  endif()
endif()
add_library(smh_q_shared SHARED src/ring.cpp bindings/c_api.cpp)
target_include_directories(smh_q_shared PUBLIC include)
set_target_properties(smh_q_shared PROPERTIES OUTPUT_NAME smh_q)
function(add_smh_example name)
  add_executable(${name} examples/${name}.cpp)
  target_link_libraries(${name} PRIVATE smh_q)
endfunction()
function(add_smh_benchmark name)
  add_executable(${name} benchmarks/${name}.cpp)
  target_link_libraries(${name} PRIVATE smh_q)
  if(SMH_Q_NATIVE_ARCH)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(${name} PRIVATE -march=native)
    endif()
  endif()
endfunction()
add_smh_example(roundtrip)
add_smh_example(producer)
add_smh_example(consumer)
add_smh_example(stress)
add_smh_benchmark(bench_throughput)
add_smh_benchmark(bench_sequential)
add_smh_benchmark(bench_futex)
enable_testing()
add_test(NAME cpp_roundtrip COMMAND roundtrip)
add_test(NAME cpp_stress COMMAND stress)
option(SMH_Q_BUILD_PYTHON "Build pybind11 module" OFF)
if(SMH_Q_BUILD_PYTHON)
  include(FetchContent)
  FetchContent_Declare(pybind11 GIT_REPOSITORY https://github.com/pybind/pybind11.git GIT_TAG v2.13.6)
  FetchContent_MakeAvailable(pybind11)
  pybind11_add_module(_native MODULE bindings/pybind_module.cpp)
  target_link_libraries(_native PRIVATE smh_q)
  target_include_directories(_native PRIVATE include)
  install(TARGETS _native LIBRARY DESTINATION smh_q)
endif()
